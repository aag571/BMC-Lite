cmake_minimum_required(VERSION 3.20)
project(bmc_lite VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BMC_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(BMC_TSAN "Enable ThreadSanitizer" OFF)
find_package(Threads REQUIRED)
add_library(bmc_core src/sel.cpp src/state.cpp src/event_bus.cpp src/fd.cpp
  src/config.cpp src/rules.cpp src/recovery.cpp src/engine.cpp src/readers.cpp
  src/logger.cpp src/worker.cpp src/actions.cpp src/monitor.cpp src/linux_io.cpp)
target_include_directories(bmc_core PUBLIC include)
target_link_libraries(bmc_core PUBLIC Threads::Threads)
target_compile_options(bmc_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(BMC_TSAN AND BMC_SANITIZE)
  message(FATAL_ERROR "BMC_TSAN and BMC_SANITIZE require separate builds")
endif()
if(BMC_TSAN)
  target_compile_options(bmc_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(bmc_core PUBLIC -fsanitize=thread)
endif()
if(BMC_SANITIZE)
  target_compile_options(bmc_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(bmc_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(bmc-lite src/main.cpp)
target_link_libraries(bmc-lite PRIVATE bmc_core)
target_compile_options(bmc-lite PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
include(CTest)
if(BUILD_TESTING)
  include(FetchContent)
  FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(googletest)
  add_executable(bmc_tests tests/core_test.cpp)
  target_link_libraries(bmc_tests PRIVATE bmc_core GTest::gtest_main)
  # 测试直接使用 bmc/action.hpp 与 bmc/linux_io.hpp，显式声明而非依赖传递包含。
  target_include_directories(bmc_tests PRIVATE include)
  include(GoogleTest)
  gtest_discover_tests(bmc_tests)
  add_executable(bmc_stress tests/stress_test.cpp)
  target_link_libraries(bmc_stress PRIVATE bmc_core)
  add_test(NAME scheduler_stress COMMAND bmc_stress)
  set_tests_properties(scheduler_stress PROPERTIES TIMEOUT 60 LABELS stress)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME runtime_reload COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_test.py $<TARGET_FILE:bmc-lite>)
    set_tests_properties(runtime_reload PROPERTIES TIMEOUT 20)
    add_test(NAME metrics_export COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/metrics_test.py)
    add_test(NAME fault_injection COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/fault_injection_test.py $<TARGET_FILE:bmc-lite>)
    add_test(NAME linux_io_fake COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/linux_io_fake_test.py)
    set_tests_properties(fault_injection PROPERTIES TIMEOUT 60)
  endif()
endif()
install(TARGETS bmc-lite RUNTIME DESTINATION bin)
install(FILES config/mock.conf config/hardware.conf.example config/rules.conf DESTINATION share/bmc-lite/config)
install(FILES docs/architecture.md docs/framework.md DESTINATION share/doc/bmc-lite)
